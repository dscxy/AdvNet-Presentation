\begin{frame}[t]{A relative gap makes the objective useful}
\bodyfont
For a higher-is-better metric, with reference $R$ and target $T$:
\vspace{12pt}
{\eqfont\[s=\frac{R-T}{\max(R,T)}\]}
\begin{center}
\begin{tikzpicture}[x=1pt,y=1pt]
\node[box,text width=205pt] (r) at (120,0) {Reference\\\textbf{10 Mbps}};
\node[box,text width=205pt] (t) at (440,0) {Target\\\textbf{2 Mbps}};
\node[text width=220pt,font=\fontsize{30}{38}\selectfont] at (770,0) {$\displaystyle\frac{10-2}{10}=\key{0.8}$};
\end{tikzpicture}
\end{center}
\vspace{12pt}
For a lower-is-better metric such as completion time, use $T-R$.
\takeaway{Higher score = larger target disadvantage. The score lies in $[-1,1]$.}
\end{frame}
